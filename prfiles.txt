$headers = @{
    Authorization = "Bearer <TOKEN>"
    Accept = "application/vnd.github+json"
}

$url = "https://api.github.com/search/issues?q=is:pr+is:closed+closed:>=2026-08-01+namespace+in:title&per_page=100"

$data = Invoke-RestMethod -Uri $url -Headers $headers

$data.items | Select-Object number,title,state,created_at,closed_at,html_url |
Export-Csv -Path "namespace_prs.csv" -NoTypeInformation
