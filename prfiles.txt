(async () => {
  const owner = "hcsc-paasengineering";
  const repo  = "openshift-platform-onboarding";
  const since = "2026-08-01";
  const filter = "namespace creation request";
  let page = 1, rows = [["PR","Title","Author","Created","Closed","URL"]];

  while (true) {
    const r = await fetch(`/repos/${owner}/${repo}/pulls?state=closed&per_page=100&page=${page}`,
      { headers: { "Accept": "application/vnd.github+json" }});
    const data = await r.json();
    if (!Array.isArray(data) || data.length === 0) break;
    for (const pr of data) {
      if (pr.closed_at && pr.closed_at.slice(0,10) >= since &&
          pr.title.toLowerCase().includes(filter.toLowerCase())) {
        rows.push([pr.number, pr.title, pr.user.login, pr.created_at, pr.closed_at, pr.html_url]);
      }
    }
    if (data.length < 100) break;
    page++;
  }

  const csv = rows.map(r => r.map(c => `"${String(c).replace(/"/g,'""')}"`).join(",")).join("\n");
  const blob = new Blob([csv], {type:"text/csv"});
  const a = document.createElement("a");
  a.href = URL.createObjectURL(blob);
  a.download = "namespace_prs.csv";
  a.click();
})();
